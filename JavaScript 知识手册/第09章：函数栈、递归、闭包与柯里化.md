# 第09章：函数栈、递归、闭包与柯里化

## 9.1 函数栈（调用栈）

**一句话解释：**
每次调用一个函数，JS 引擎就会把一个"函数执行记录"压入栈中；函数执行完毕，就从栈中弹出。这个栈就是**调用栈（Call Stack）**。

**餐厅比喻：**
厨房的**配菜单**——每来一张新订单，就**叠在最上面**（压栈）；做好一道菜，就把最上面那张单子**抽走**（弹出）。始终是**最后放上去的最先被处理**。

~~~javascript
// ========== 调用栈演示 ==========
function 做汤() {
  console.log("做汤完成");
}

function 炒菜() {
  做汤();                  // 做汤入栈
  console.log("炒菜完成");  // 做汤弹出后，炒菜继续
}

function 接单() {
  炒菜();                  // 炒菜入栈
  console.log("接单完成");  // 炒菜弹出后，接单继续
}

接单();

// 执行过程：
// 1. 接单()        → 调用栈：[接单]
// 2. 炒菜()        → 调用栈：[接单, 炒菜]
// 3. 做汤()        → 调用栈：[接单, 炒菜, 做汤]
// 4. 做汤完成      → 调用栈：[接单, 炒菜]（做汤弹出）
// 5. 炒菜完成      → 调用栈：[接单]（炒菜弹出）
// 6. 接单完成      → 调用栈：[]（接单弹出，栈空了）
~~~

### 调用栈图解

~~~
时间 →

执行顺序：接单 → 炒菜 → 做汤 → (做汤完成) → (炒菜完成) → (接单完成)

  调用栈变化：

  [接单]                  ← 接单入栈
  [接单, 炒菜]            ← 炒菜入栈
  [接单, 炒菜, 做汤]      ← 做汤入栈（栈顶 = 当前执行的函数）
  [接单, 炒菜]            ← 做汤完成，弹出
  [接单]                  ← 炒菜完成，弹出
  []                      ← 接单完成，弹出（栈空）
~~~

## 9.2 栈溢出（Stack Overflow）

**一句话解释：**
调用栈有大小限制。如果函数嵌套调用太深（比如没有正确终止条件的递归），栈空间被撑爆，就会报 **RangeError: Maximum call stack size exceeded**。

**餐厅比喻：**
配菜单叠得太高了——厨房的盘子架只能放 100 张单子，你叠了 10000 张，架子塌了。

~~~javascript
// ========== 栈溢出演示 ==========
function 永远不停() {
  永远不停();  // 无限递归，调用栈无限增长
}

// 永远不停();
// RangeError: Maximum call stack size exceeded

// ========== 有终止条件就不会溢出 ==========
let 计数 = 0;
function 数到100() {
  计数++;
  if (计数 >= 100) return;  // 终止条件
  数到100();
}
数到100();
console.log(计数);  // 100（正常完成，没有溢出）
~~~

## 9.3 递归

**一句话解释：**
函数**调用自己**。必须有**终止条件**（基准情况），否则会无限递归导致栈溢出。

**餐厅比喻：**
你问服务员"这道菜怎么做？"，服务员说"你问厨师"。厨师说"你问帮厨"。帮厨说"先把肉焯水"——**最底层的人给出最终答案，然后一层层传回来**。

### 阶乘

~~~javascript
// 5! = 5 × 4 × 3 × 2 × 1 = 120
function 阶乘(n) {
  // 基准情况（终止条件）
  if (n <= 1) return 1;
  // 递归情况（调用自己）
  return n * 阶乘(n - 1);
}

console.log(阶乘(5));  // 120

// 执行过程：
// 阶乘(5) → 5 × 阶乘(4)
// 阶乘(4) → 4 × 阶乘(3)
// 阶乘(3) → 3 × 阶乘(2)
// 阶乘(2) → 2 × 阶乘(1)
// 阶乘(1) → 1（基准情况，不再递归）
//
// 然后一层层返回：
// 阶乘(2) = 2 × 1 = 2
// 阶乘(3) = 3 × 2 = 6
// 阶乘(4) = 4 × 6 = 24
// 阶乘(5) = 5 × 24 = 120
~~~

### 斐波那契数列

~~~javascript
// 1, 1, 2, 3, 5, 8, 13, 21, ...
// 规则：第 n 个数 = 第(n-1)个数 + 第(n-2)个数
function 斐波那契(n) {
  if (n <= 2) return 1;           // 基准情况
  return 斐波那契(n - 1) + 斐波那契(n - 2);  // 递归情况
}

console.log(斐波那契(1));   // 1
console.log(斐波那契(6));   // 8
console.log(斐波那契(10));  // 55
~~~

### 递归的实际应用

~~~javascript
// ========== 递归求和 ==========
function 求和(数组, 索引 = 0) {
  if (索引 >= 数组.length) return 0;
  return 数组[索引] + 求和(数组, 索引 + 1);
}
console.log(求和([1, 2, 3, 4, 5]));  // 15

// ========== 递归深拷贝 ==========
function 深拷贝(对象) {
  if (对象 === null || typeof 对象 !== "object") return 对象;
  if (Array.isArray(对象)) return 对象.map(项 => 深拷贝(项));
  const 拷贝 = {};
  for (const 键 of Object.keys(对象)) {
    拷贝[键] = 深拷贝(对象[键]);
  }
  return 拷贝;
}

const 原始 = { 菜名: "红烧肉", 配料: ["五花肉", "酱油"], 详情: { 口味: "甜" } };
const 拷贝 = 深拷贝(原始);

// ========== 递归遍历树形结构 ==========
const 组织架构 = {

      { 名字: "服务员A", 下属: [] },

    ]}
  ]
};

function 打印组织(节点, 层级 = 0) {
  console.log("  ".repeat(层级) + 节点.名字);
  for (const 下属 of 节点.下属) {
    打印组织(下属, 层级 + 1);
  }
}

打印组织(组织架构);
// 总经理
//   厨师长
//     张师傅
//     李师傅
//   店长
//     服务员A
//     服务员B
~~~

### 记忆化（Memoization）

~~~javascript
// ========== 问题：普通递归斐波那契有大量重复计算 ==========
let 计算次数 = 0;
function 慢斐波那契(n) {
  计算次数++;
  if (n <= 2) return 1;
  return 慢斐波那契(n - 1) + 慢斐波那契(n - 2);
}
慢斐波那契(40);
console.log(`计算了${计算次数}次`);  // 计算了331160777次！太慢了

// ========== 解决：记忆化——用缓存记住已经算过的结果 ==========
let 快计算次数 = 0;
function 创建记忆化() {
  const 缓存 = {};

  return function 斐波那契(n) {
    快计算次数++;

    if (n in 缓存) return 缓存[n];  // 已经算过了，直接返回缓存
    if (n <= 2) return 1;

    缓存[n] = 斐波那契(n - 1) + 斐波那契(n - 2);  // 算完存进缓存
    return 缓存[n];
  };
}

const 快斐波那契 = 创建记忆化();
console.log(快斐波那契(40));  // 102334155
console.log(`计算了${快计算次数}次`);  // 计算了77次！快了几百万倍

// ========== 通用记忆化函数 ==========
function 记忆化(fn) {
  const 缓存 = {};
  return function(...参数) {
    const 键 = JSON.stringify(参数);
    if (缓存[键] !== undefined) return 缓存[键];
    缓存[键] = fn.apply(this, 参数);
    return 缓存[键];
  };
}

const 记忆化阶乘 = 记忆化(function(n) {
  if (n <= 1) return 1;
  return n * 记忆化阶乘(n - 1);
});

console.log(记忆化阶乘(10));  // 3628800
console.log(记忆化阶乘(10));  // 3628800（第二次直接从缓存取，瞬间完成）
~~~

## 9.4 闭包

**一句话解释：**
当一个**内部函数**在**外部被调用**时，它仍然能访问到**它被创建时所在函数的变量**——即使那个外层函数已经执行完毕。这种"内部函数记住了外部环境"的特性就是**闭包（Closure）**。

**餐厅比喻：**
你去餐厅吃饭，厨师给你做了一道菜，顺便**把他家酱料的配方写在纸条上给你了**。你回到家（外层函数已经执行完了），拿出纸条（闭包），还是能按配方调出酱料（访问外层变量）。**纸条记住了酱料配方，这就是闭包。**

### 核心概念

~~~javascript
// ========== 闭包的基本结构 ==========
function 厨房() {
  const 秘制酱料 = "酱油+醋+蒜+辣椒油";  // 外层函数的变量

  // 内部函数（闭包）
  function 做蘸料() {
    console.log(`蘸料配方：${秘制酱料}`);  // 内部函数访问外部变量
  }

  return 做蘸料;  // 把内部函数返回出去
}

const 我的蘸料 = 厨房();  // 厨房执行完毕，秘制酱料理论上该"消失"了
我的蘸料();              // 蘸料配方：酱油+醋+蒜+辣椒油（还能访问！）

// 闭包的关键：
// 1. 函数嵌套（内部函数在外部函数里面）
// 2. 内部函数引用了外部函数的变量
// 3. 内部函数在外部被使用（返回或传出）
// 4. 外部函数执行完毕后，内部函数仍然能访问那些变量
~~~

### 私有变量

~~~javascript
// ========== 闭包实现私有变量 ==========
function 创建收银台(初始余额) {
  let 余额 = 初始余额;  // 私有变量：外部无法直接访问

  return {
    存款(金额) {
      余额 += 金额;
      console.log(`存入${金额}，余额：${余额}`);
    },
    取款(金额) {
      if (金额 > 余额) {
        console.log("余额不足！");
        return;
      }
      余额 -= 金额;
      console.log(`取出${金额}，余额：${余额}`);
    },
    查余额() {
      console.log(`当前余额：${余额}`);
    }
  };
}

const 收银台 = 创建收银台(1000);
收银台.查余额();      // 当前余额：1000
收银台.存款(500);     // 存入500，余额：1500
收银台.取款(200);     // 取出200，余额：1300
收银台.查余额();      // 当前余额：1300

// 无法直接访问余额
console.log(收银台.余额);  // undefined（私有变量，外部访问不到）
// console.log(余额);      // ReferenceError

// 但可以创建多个独立的收银台
const 收银台2 = 创建收银台(500);
收银台2.查余额();  // 当前余额：500（独立的，不受收银台影响）
~~~

### 防抖（Debounce）

~~~javascript
// ========== 防抖：用户停止操作后才执行 ==========
function 创建防抖(回调, 延迟) {
  let 定时器;  // 闭包变量：定时器ID

  return function(...参数) {
    clearTimeout(定时器);       // 每次调用都清除上一次的定时器
    定时器 = setTimeout(() => {
      回调.apply(this, 参数);   // 延迟后执行
    }, 延迟);
  };
}

// 用户停止输入 500ms 后才发搜索请求
const 防抖搜索 = 创建防抖(function(关键词) {
  console.log(`搜索：${关键词}`);
}, 500);

// 模拟用户快速输入
防抖搜索("红");
防抖搜索("红烧");
防抖搜索("红烧肉");
// 只会执行最后一次："搜索：红烧肉"
~~~

### 缓存（Cache）

~~~javascript
// ========== 闭包实现缓存 ==========
function 创建缓存函数(fn) {
  const 缓存 = {};  // 闭包变量：缓存对象

  return function(参数) {
    if (参数 in 缓存) {
      console.log(`[缓存命中] ${参数}`);
      return 缓存[参数];
    }
    console.log(`[计算中] ${参数}`);
    缓存[参数] = fn(参数);
    return 缓存[参数];
  };
}

const 计算价格 = 创建缓存函数(function(菜品) {
  // 假装这是一个耗时计算
  return 菜品.length * 10;
});

console.log(计算价格("红烧肉"));    // [计算中] 红烧肉 → 30
console.log(计算价格("红烧肉"));    // [缓存命中] 红烧肉 → 30（不重新计算）
console.log(计算价格("麻婆豆腐"));  // [计算中] 麻婆豆腐 → 40
~~~

### 闭包的循环陷阱

~~~javascript
// ========== 经典陷阱：var 的循环 ==========
// ❌ 问题代码
for (var i = 0; i < 3; i++) {
  setTimeout(() => {
    console.log(i);
  }, 100);
}
// 输出：3 3 3（不是 0 1 2）
// 原因：var 没有块级作用域，i 是同一个变量，循环结束后 i = 3

// ✅ 解决1：用 let（推荐）
for (let i = 0; i < 3; i++) {
  setTimeout(() => {
    console.log(i);
  }, 100);
}
// 输出：0 1 2（let 有块级作用域，每次循环创建新的 i）

// ✅ 解决2：用 IIFE 创建闭包
for (var i = 0; i < 3; i++) {
  (function(当前i) {
    setTimeout(() => {
      console.log(当前i);
    }, 100);
  })(i);  // 把当前的 i 传进去
}
// 输出：0 1 2

// ✅ 解决3：用 bind
for (var i = 0; i < 3; i++) {
  setTimeout(console.log.bind(null, i), 100);
}
// 输出：0 1 2
~~~

~~~javascript
function 创建大数据() {
~~~

  const 秘制酱料 = "酱油+醋+蒜";   ← 步骤1：创建变量

// └──────────────────────────────────────────┘
//                 ↑ 闭包引用
// ┌──────────────────────────────────────────┐
// │  做蘸料函数（闭包）                       │
// │  [[Scope]] = 秘制酱料                     │
// └──────────────────────────────────────────┘
~~~

## 9.5 Vue Hook 与闭包

**一句话解释：**
Vue 3 的 Composition API 中的 `ref`、`computed`、`watch` 等 hook，本质上都是**利用闭包**把响应式数据"锁"在闭包环境里，让组件函数多次执行时能**访问到同一个数据**。

**餐厅比喻：**
Vue 组件就像一个**每天早上重新开张的餐厅**——每天都要重新执行一遍 setup 里的代码。但 `ref` 创建的响应式数据就像一个**保险箱**（闭包），每天餐厅重新开张时，保险箱里的东西还在，不会丢。

~~~javascript
// ========== Vue 3 组件（简化模拟） ==========
function 模拟Vue组件() {
  // setup 部分——组件初始化时执行一次

  // ref 创建响应式数据 → 本质是闭包
  function ref(初始值) {
    let 值 = 初始值;  // 闭包变量

    return {
      get value() { return 值; },
      set value(新值) { 值 = 新值; }
    };
  }

  // 这个计数器是一个闭包
  const 计数 = ref(0);  // 计数.value 被闭包"锁住"了

  function 增加() {
    计数.value++;  // 每次调用都访问同一个闭包里的值
    console.log(`计数：${计数.value}`);
  }

  // 模拟多次调用——每次调用增加() 都能访问同一个计数
  增加();  // 计数：1
  增加();  // 计数：2
  增加();  // 计数：3

  // 即使 setup() 执行完毕，计数 仍然存在于闭包中
  return { 计数, 增加 };
}

const 组件 = 模拟Vue组件();
组件.增加();  // 计数：4（继续访问同一个闭包变量）
组件.增加();  // 计数：5
~~~

~~~javascript
// ========== computed 也是闭包 ==========
function computed(fn) {
  let 缓存值;
  let 已过期 = true;  // 闭包变量

  return {
    get value() {
      if (已过期) {
        缓存值 = fn();  // 重新计算
        已过期 = false;
      }
      return 缓存值;  // 未过期则返回缓存
    },
    标记过期() {
      已过期 = true;
    }
  };
}

// ========== watch 也是闭包 ==========
function watch(source, callback) {
  let 旧值 = source();  // 闭包变量记住旧值

  return function 检查变化() {
    const 新值 = source();
    if (新值 !== 旧值) {
      callback(新值, 旧值);  // 值变了才回调
      旧值 = 新值;  // 更新旧值
    }
  };
}
~~~

## 9.6 柯里化（Currying）

**一句话解释：**
柯里化是把一个**接收多个参数的函数**，改造成一系列**每次只接收一个参数的函数**。每传一个参数，就返回一个**新函数**，等着接收下一个参数，直到所有参数都齐了才执行。

**餐厅比喻：**
你去咖啡店点单——
- **普通函数**：你一次性说"大杯、热、拿铁、加糖"，服务员一次性全部记下来
- **柯里化**：服务员一个问题一个问题问——"大杯还是小杯？"你选完，"热的还是冰的？"你选完，"什么口味？"你选完，"加不加糖？"你选完。**每回答一个问题，就返回一个等你回答下一个问题的"状态"**。

### 基本概念

~~~javascript
// ========== 普通函数：一次性传所有参数 ==========
function 做菜(调料, 食材, 份量) {
  return `${调料}${食材}，${份量}份`;
}

console.log(做菜("红烧", "肉", 2));  // 红烧肉，2份

// ========== 柯里化：一次传一个参数 ==========
function 柯里化做菜(调料) {
  return function(食材) {
    return function(份量) {
      return `${调料}${食材}，${份量}份`;
    };
  };
}

console.log(柯里化做菜("红烧")("肉")(2));  // 红烧肉，2份

// 也可以分步调用
const 红烧 = 柯里化做菜("红烧");
const 红烧肉 = 红烧("肉");
console.log(红烧肉(1));  // 红烧肉，1份
console.log(红烧肉(3));  // 红烧肉，3份

// 换一种调料
const 糖醋 = 柯里化做菜("糖醋");
const 糖醋里脊 = 糖醋("里脊");
console.log(糖醋里脊(2));  // 糖醋里脊，2份
~~~

### 手写柯里化函数

~~~javascript
// ========== 手写通用柯里化函数 ==========
function 柯里化(fn) {
  return function 递归收集(...已收集的参数) {
    // 如果收集的参数数量足够了，就执行原函数
    if (已收集的参数.length >= fn.length) {
      return fn(...已收集的参数);
    }
    // 否则返回一个新函数，继续收集参数
    return function(...更多参数) {
      return 递归收集(...已收集的参数, ...更多参数);
    };
  };
}

// ========== 逐步拆解过程 ==========

// 假设原函数需要3个参数
function 做菜(调料, 食材, 份量) {
  return `${调料}${食材}，${份量}份`;
}
console.log(做菜.length);  // 3（形参个数）

const 柯里做菜 = 柯里化(做菜);

// 第一次调用：传了1个参数，需要3个，不够
const 步骤1 = 柯里做菜("红烧");
// → 返回递归收集(已收集=["红烧"])
// → 已收集.length(1) < fn.length(3) → 返回新函数

// 第二次调用：又传了1个，总共2个，还是不够
const 步骤2 = 步骤1("肉");
// → 返回递归收集(已收集=["红烧", "肉"])
// → 已收集.length(2) < fn.length(3) → 返回新函数

// 第三次调用：又传了1个，总共3个，够了！
const 结果 = 步骤2(2);
// → 递归收集(已收集=["红烧", "肉", 2])
// → 已收集.length(3) >= fn.length(3) → 执行原函数
// → 做菜("红烧", "肉", 2) → "红烧肉，2份"

console.log(结果);  // 红烧肉，2份

// 也可以一次性传完
console.log(柯里做菜("糖醋")("里脊")(3));  // 糖醋里脊，3份
console.log(柯里做菜("糖醋", "里脊")(3));   // 糖醋里脊，3份（前两个一起传）
console.log(柯里做菜("糖醋", "里脊", 3));   // 糖醋里脊，3份（全部一起传）
~~~

### 实际应用

~~~javascript
// ========== 应用1：批量创建半成品函数 ==========
const 柯里化做菜 = 柯里化(function(调料, 食材, 份量) {
  return `${调料}${食材}，${份量}份`;
});

// 预设"红烧"，创建一批红烧菜的函数
const 红烧系列 = 柯里化做菜("红烧");
console.log(红烧系列("肉")(1));      // 红烧肉，1份
console.log(红烧系列("排骨")(2));    // 红烧排骨，2份
console.log(红烧系列("鱼")(1));      // 红烧鱼，1份

// 预设"糖醋"
const 糖醋系列 = 柯里化做菜("糖醋");
console.log(糖醋系列("里脊")(1));    // 糖醋里脊，1份
console.log(糖醋系列("排骨")(2));    // 糖醋排骨，2份

// ========== 应用2：配合数组方法 ==========
const 柯里化筛选 = 柯里化(function(条件函数, 数组) {
  return 数组.filter(条件函数);
});

const 菜单 = [
  { 名字: "红烧肉", 价格: 38 },
  { 名字: "麻婆豆腐", 价格: 22 },
  { 名字: "糖醋里脊", 价格: 45 },
  { 名字: "拍黄瓜", 价格: 12 }
];

// 创建"按价格筛选"的半成品函数
const 筛选 = 柯里化筛选((菜) => 菜.价格 > 30);
console.log(筛选(菜单));
// [{ 名字: "红烧肉", 价格: 38 }, { 名字: "糖醋里脊", 价格: 45 }]

// ========== 应用3：日志系统 ==========
const 柯里化日志 = 柯里化(function(级别, 模块, 消息) {
  console.log(`[${级别}][${模块}] ${消息}`);
});

const 信息日志 = 柯里化日志("INFO");
const 警告日志 = 柯里化日志("WARN");
const 错误日志 = 柯里化日志("ERROR");

const 厨房信息 = 信息日志("厨房");
const 前厅警告 = 警告日志("前厅");

厨房信息("红烧肉开始炖了");     // [INFO][厨房] 红烧肉开始炖了
厨房信息("麻婆豆腐出锅了");     // [INFO][厨房] 麻婆豆腐出锅了
前厅警告("3号桌等太久了");       // [WARN][前厅] 3号桌等太久了
错误日志("厨房")("着火了！！");  // [ERROR][厨房] 着火了！！
~~~

### 柯里化 vs 偏函数（Partial Application）

~~~javascript
// 柯里化：每次只接收一个参数，返回新函数
// 偏函数：一次可以预设多个参数，返回新函数

// ========== 柯里化 ==========
function 柯里化做菜(调料) {
  return function(食材) {
    return function(份量) {
      return `${调料}${食材}，${份量}份`;
    };
  };
}
// 调用：柯里化做菜("红烧")("肉")(2)  → 每次一个参数

// ========== 偏函数 ==========
function 偏函数做菜(调料, 食材) {
  return function(份量) {
    return `${调料}${食材}，${份量}份`;
  };
}
// 调用：偏函数做菜("红烧", "肉")(2)  → 前面可以一次传多个

// 区别：
// 柯里化：自动拆成一次一个参数的链式调用
// 偏函数：手动预设部分参数，返回一个接收剩余参数的函数
~~~

## 第09章 总结

| 术语 | 一句话解释 | 餐厅比喻关键词 |
|---|---|---|
| 调用栈 | 函数调用时压栈，执行完弹栈 | 配菜单叠起来，做好就抽走 |
| 栈溢出 | 调用栈被撑爆 | 盘子架上单子叠太高，塌了 |
| 递归 | 函数调用自己 | 问服务员→问厨师→问帮厨→给出答案 |
| 基准情况 | 递归的终止条件 | 帮厨说"先把肉焯水"（不再继续问） |
| 记忆化 | 用缓存避免重复计算 | 做过的菜把配方记下来，下次不重新研究 |
| 闭包 | 内部函数"记住"外部环境 | 厨师给你酱料配方纸条，带回家还能用 |
| 私有变量 | 闭包保护变量不被外部直接访问 | 保险箱里的钱，外面看不到但能通过接口操作 |
| 函数工厂 | 闭包预设参数，批量创建函数 | 预设"红烧"，批量出红烧系列菜 |
| 防抖 | 停止操作后才执行 | 停止点菜5秒后才通知厨房 |
| 缓存 | 闭包记住已计算的结果 | 做过的菜配方记下来，直接用 |
| 循环陷阱 | var 的闭包问题 | 每道菜都用了同一个变量，结果全一样 |
| Vue Hook 闭包 | ref/computed/watch 都是闭包 | 保险箱锁住响应式数据，重新开张数据还在 |
| 柯里化 | 多参数函数变成单参数链式调用 | 服务员一个问题一个问题问 |
| 手写柯里化 | 判断参数够不够，不够返回新函数 | 问完一个问题，再问下一个 |
| 偏函数 | 预设部分参数，返回新函数 | 一次预设两个参数，剩下再问 |
