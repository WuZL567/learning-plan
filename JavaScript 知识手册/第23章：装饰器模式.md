# 第23章：装饰器模式

## 23.1 装饰器模式的概念

**一句话解释：**
装饰器模式是一种**不修改原有函数或类**的情况下，给它们**增加新功能**的设计模式。就像给一个东西"套一层外衣"——东西本身没变，但穿上外衣后多了新能力。

**餐厅比喻：**
你有一道红烧肉（原始函数），你想给它**加一个"上菜时拍照发朋友圈"的功能**。你不需要改红烧肉的做法，只需要在上菜环节**额外加一步拍照**。装饰器就是这个"额外加的一步"——它**包裹**在原始函数外面，不动内部。

~~~
装饰器的工作方式：

原始函数：   做菜()
装饰后：     拍照(做菜())  ← 在做菜前后各加了逻辑

┌─────────────────────────────────┐
│  装饰器（外层包装）              │
│  ┌───────────────────────────┐  │
│  │  前置逻辑：拍照、记录日志   │  │
│  │  ┌─────────────────────┐  │  │
│  │  │  原始函数：做菜()    │  │  │
│  │  │  （没被修改）        │  │  │
│  │  └─────────────────────┘  │  │
│  │  后置逻辑：通知上菜、收钱  │  │
│  └───────────────────────────┘  │
└─────────────────────────────────┘
~~~

## 23.2 函数装饰器（高阶函数实现）

**一句话解释：**
用一个**高阶函数**（接收函数、返回新函数）包裹原函数，在原函数执行前后插入额外逻辑。

~~~javascript
// ========== 最基本的函数装饰器 ==========

// 原始函数
function 做菜(菜名) {
  console.log(`正在做${菜名}...`);
  return `${菜名}做好了`;
}

// 装饰器：给任何函数加日志功能
function 日志装饰器(原函数) {
  return function(...参数) {
    console.log(`[日志] 调用 ${原函数.name}，参数：${参数}`);
    const 开始时间 = Date.now();

    const 结果 = 原函数.apply(this, 参数);  // 执行原函数

    const 耗时 = Date.now() - 开始时间;
    console.log(`[日志] ${原函数.name} 完成，耗时：${耗时}ms，结果：${结果}`);

    return 结果;
  };
}

// 装饰
const 装饰后的做菜 = 日志装饰器(做菜);

装饰后的做菜("红烧肉");
// [日志] 调用 做菜，参数：红烧肉
// 正在做红烧肉...
// [日志] 做菜 完成，耗时：0ms，结果：红烧肉做好了

// 原函数没被修改
做菜("麻婆豆腐");  // 正在做麻婆豆腐...（没有日志）
~~~

### 计时装饰器

~~~javascript
// ========== 计时装饰器 ==========
function 计时装饰器(原函数) {
  return function(...参数) {
    const 开始 = performance.now();
    const 结果 = 原函数.apply(this, 参数);
    const 结束 = performance.now();
    console.log(`⏱️ ${原函数.name} 耗时：${(结束 - 开始).toFixed(2)}ms`);
    return 结果;
  };
}

function 复杂计算(n) {
  let 总和 = 0;
  for (let i = 0; i < n; i++) {
    总和 += Math.sqrt(i);
  }
  return 总和;
}

const 计时计算 = 计时装饰器(复杂计算);
计时计算(1000000);  // ⏱️ 复杂计算 耗时：12.34ms
~~~

### 权限检查装饰器

~~~javascript
// ========== 权限检查装饰器 ==========
function 需要权限(所需权限) {
  return function(原函数) {
    return function(...参数) {
      const 当前用户 = 获取当前用户();

      if (!当前用户.权限.includes(所需权限)) {
        console.log(`❌ 权限不足：需要"${所需权限}"权限`);
        return null;
      }

      console.log(`✅ 权限验证通过`);
      return 原函数.apply(this, 参数);
    };
  };
}

function 获取当前用户() {
  return { 名字: "张师傅", 权限: ["做菜", "采购"] };
}

function 修改菜单(新菜单) {
  console.log(`菜单已更新：${新菜单}`);
}

function 查看账单() {
  console.log("今日流水：8500元");
}

// 用装饰器包装
const 安全修改菜单 = 需要权限("管理员")(修改菜单);
const 安全查看账单 = 需要权限("做菜")(查看账单);

安全修改菜单("新菜单");  // ❌ 权限不足：需要"管理员"权限
安全查看账单();           // ✅ 权限验证通过 → 今日流水：8500元
~~~

### 缓存装饰器（记忆化）

~~~javascript
// ========== 缓存装饰器 ==========
function 缓存装饰器(原函数) {
  const 缓存 = {};

  return function(...参数) {
    const 键 = JSON.stringify(参数);

    if (缓存[键] !== undefined) {
      console.log(`[缓存命中] ${原函数.name}(${参数})`);
      return 缓存[键];
    }

    console.log(`[计算中] ${原函数.name}(${参数})`);
    缓存[键] = 原函数.apply(this, 参数);
    return 缓存[键];
  };
}

function 计算价格(菜品, 数量) {
  // 假装这是个复杂计算
  const 价格表 = { "红烧肉": 38, "麻婆豆腐": 22 };
  return (价格表[菜品] || 0) * 数量;
}

const 缓存计算价格 = 缓存装饰器(计算价格);

缓存计算价格("红烧肉", 2);  // [计算中] 计算价格(红烧肉,2) → 76
缓存计算价格("红烧肉", 2);  // [缓存命中] 计算价格(红烧肉,2) → 76（瞬间返回）
缓存计算价格("麻婆豆腐", 1); // [计算中] 计算价格(麻婆豆腐,1) → 22
~~~

### 重试装饰器

~~~javascript
// ========== 重试装饰器 ==========
function 重试装饰器(最大重试次数 = 3, 延迟 = 1000) {
  return function(原函数) {
    return async function(...参数) {
      for (let 尝试次数 = 1; 尝试次数 <= 最大重试次数; 尝试次数++) {
        try {
          return await 原函数.apply(this, 参数);
        } catch (错误) {
          console.log(`第${尝试次数}次尝试失败：${错误.message}`);
          if (尝试次数 === 最大重试次数) {
            throw 错误;  // 最后一次也失败了，抛出错误
          }
          console.log(`${延迟}ms 后重试...`);
          await new Promise(r => setTimeout(r, 延迟));
        }
      }
    };
  };
}

// 模拟一个不稳定的请求
let 请求次数 = 0;
async function 获取供应商报价(供应商) {
  请求次数++;
  if (请求次数 < 3) {
    throw new Error("网络超时");
  }
  return { 供应商, 报价: 38 };
}

const 稳定请求 = 重试装饰器(3, 500)(获取供应商报价);
// 稳定请求("供应商A").then(console.log);
// 第1次尝试失败：网络超时 → 500ms后重试
// 第2次尝试失败：网络超时 → 500ms后重试
// → { 供应商: "供应商A", 报价: 38 }
~~~

### 装饰器组合（链式装饰）

~~~javascript
// ========== 多个装饰器叠加 ==========
function 组合装饰器(...装饰器们) {
  return function(原函数) {
    // 从右到左依次装饰
    return 装饰器们.reduce(
      (被装饰的函数, 装饰器) => 装饰器(被装饰的函数),
      原函数
    );
  };
}

// 定义多个装饰器
function 添加日志(原函数) {
  return function(...参数) {
    console.log(`[日志] 调用 ${原函数.name}`);
    return 原函数.apply(this, 参数);
  };
}

function 添加计时(原函数) {
  return function(...参数) {
    const 开始 = Date.now();
    const 结果 = 原函数.apply(this, 参数);
    console.log(`[计时] ${Date.now() - 开始}ms`);
    return 结果;
  };
}

function 添加缓存(原函数) {
  const 缓存 = {};
  return function(参数) {
    if (缓存[参数]) return 缓存[参数];
    缓存[参数] = 原函数.call(this, 参数);
    return 缓存[参数];
  };
}

function 做菜(菜名) {
  console.log(`做${菜名}`);
  return `${菜名}做好了`;
}

// 组合装饰：做菜 → 加缓存 → 加计时 → 加日志
const 超级做菜 = 组合装饰器(添加日志, 添加计时, 添加缓存)(做菜);

超级做菜("红烧肉");
// [日志] 调用 做菜
// 做红烧肉
// [计时] 0ms
// 红烧肉做好了
~~~

## 23.3 方法装饰器

**一句话解释：**
装饰类的某个方法，在不修改原方法的情况下增加功能（日志、权限、缓存、防抖等）。

~~~javascript
// ========== 给类的方法加装饰器（手动方式） ==========

class 餐厅 {
  constructor(名字) {
    this.名字 = 名字;
  }

  点菜(菜名) {
    console.log(`${this.名字}开始做${菜名}`);
    return `${菜名}做好了`;
  }

  结账(金额) {
    console.log(`收了${金额}元`);
    return 金额;
  }
}

// 手动装饰某个方法
const 原始点菜 = 餐厅.prototype.点菜;

餐厅.prototype.点菜 = function(...参数) {
  console.log(`[装饰] 点菜前：检查食材`);
  const 结果 = 原始点菜.apply(this, 参数);
  console.log(`[装饰] 点菜后：通知服务员上菜`);
  return 结果;
};

const 店 = new 餐厅("老王饭馆");
店.点菜("红烧肉");
// [装饰] 点菜前：检查食材
// 老王饭馆开始做红烧肉
// [装饰] 点菜后：通知服务员上菜

// ========== 防抖装饰器（用于搜索等场景） ==========
function 防抖装饰器(延迟) {
  return function(原函数) {
    let 定时器;
    return function(...参数) {
      clearTimeout(定时器);
      定时器 = setTimeout(() => {
        原函数.apply(this, 参数);
      }, 延迟);
    };
  };
}

class 搜索服务 {
  constructor() {
    // 把搜索方法装饰成防抖版本
    this.搜索 = 防抖装饰器(300)(this.搜索.bind(this));
  }

  搜索(关键词) {
    console.log(`搜索：${关键词}`);
  }
}

const 搜索 = new 搜索服务();
搜索.搜索("红");
搜索.搜索("红烧");
搜索.搜索("红烧肉");
// 只会执行最后一次："搜索：红烧肉"

// ========== 节流装饰器 ==========
function 节流装饰器(间隔) {
  return function(原函数) {
    let 上次执行 = 0;
    return function(...参数) {
      const 现在 = Date.now();
      if (现在 - 上次执行 >= 间隔) {
        上次执行 = 现在;
        return 原函数.apply(this, 参数);
      }
    };
  };
}
~~~

## 23.4 类装饰器

**一句话解释：**
装饰整个类——在类创建时包裹一层，可以修改类的行为、添加属性、甚至替换整个类。

~~~javascript
// ========== 类装饰器：给类添加静态属性 ==========
function 添加版本(版本号) {
  return function(原类) {
    原类.版本 = 版本号;
    return 原类;
  };
}

// 手动应用装饰器
class 餐厅 {
  constructor(名字) {
    this.名字 = 名字;
  }
}
餐厅 = 添加版本("1.0.0")(餐厅);

console.log(餐厅.版本);  // 1.0.0

// ========== 类装饰器：自动注册 ==========
const 全局注册表 = {};

function 自动注册(名称) {
  return function(原类) {
    全局注册表[名称] = 原类;
    return 原类;
  };
}

// 手动装饰
class 红烧肉 { }
红烧肉 = 自动注册("红烧肉")(红烧肉);

class 麻婆豆腐 { }
麻婆豆腐 = 自动注册("麻婆豆腐")(麻婆豆腐);

console.log(Object.keys(全局注册表));  // ["红烧肉", "麻婆豆腐"]
const 菜 = new 全局注册表["红烧肉"]();

// ========== 类装饰器：混入方法 ==========
function 混入(...来源们) {
  return function(原类) {
    for (const 来源 of 来源们) {
      for (const [键, 值] of Object.entries(来源)) {
        原类.prototype[键] = 值;
      }
    }
    return 原类;
  };
}

const 打折功能 = {
  打折(折扣) {
    console.log(`开启${折扣 * 10}折优惠`);
    return this;
  }
};

const 外卖功能 = {
  开通外卖(平台) {
    console.log(`在${平台}上线`);
    return this;
  }
};

class 餐厅2 { }
餐厅2 = 混入(打折功能, 外卖功能)(餐厅2);

const 店 = new 餐厅2();
店.打折(0.8);           // 开启8折优惠
店.开通外卖("美团");    // 在美团上线
~~~

## 23.5 TC39 装饰器提案语法（@decorator）

**一句话解释：**
TC39 装饰器提案（Stage 3）提供了一种 `@装饰器名` 的语法糖，让装饰器的使用更加简洁优雅。

**餐厅比喻：**
以前的装饰方式就像"先把菜做好，再手动套一个包装盒"。`@` 语法就像**在菜谱上直接标注"上菜前拍照"**——厨师照着菜谱做，自动就会在上菜前拍照。

~~~javascript
// ========== TC39 装饰器提案语法（Stage 3） ==========

// 注意：这是最新的装饰器语法，需要转译器支持（如 TypeScript 5.0+）

// 类装饰器
function 可序列化(原类, 上下文) {
  上下文.addInitializer(function() {
    this.prototype.toJSON = function() {
      return JSON.stringify(this);
    };
  });
  return 原类;
}

@可序列化
class 菜品 {
  constructor(名字, 价格) {
    this.名字 = 名字;
    this.价格 = 价格;
  }
}

const 红烧肉 = new 菜品("红烧肉", 38);
console.log(红烧肉.toJSON());  // {"名字":"红烧肉","价格":38}

// 方法装饰器
function 日志(原方法, 上下文) {
  return function(...参数) {
    console.log(`[${上下文.name}] 被调用，参数：${参数}`);
    return 原方法.apply(this, 参数);
  };
}

class 餐厅3 {
  constructor(名字) {
    this.名字 = 名字;
  }

  @日志
  点菜(菜名) {
    return `${this.名字}做${菜名}`;
  }

  @日志
  结账(金额) {
    return `收了${金额}元`;
  }
}

const 店3 = new 餐厅3("老王饭馆");
店3.点菜("红烧肉");    // [点菜] 被调用，参数：红烧肉
店3.结账(100);          // [结账] 被调用，参数：100

// 类字段装饰器
function 默认值(值) {
  return function(初始值, 上下文) {
    return 初始值 !== undefined ? 初始值 : 值;
  };
}

class 配置 {
  @默认值("深色")
  主题;

  @默认值(16)
  字号;
}

const 配置实例 = new 配置();
console.log(配置实例.主题);  // 深色
console.log(配置实例.字号);  // 16
~~~

## 23.6 TypeScript 中的装饰器

~~~typescript
// ========== TypeScript 装饰器（实验性，需要 experimentalDecorators） ==========

// 类装饰器
function 装饰类<T extends { new (...args: any[]): {} }>(原类: T) {
  return class extends 原类 {
    版本 = "1.0.0";
    创建时间 = new Date();
  };
}

// 方法装饰器
function 装饰方法(
  目标: any,
  属性名: string,
  描述符: PropertyDescriptor
) {
  const 原方法 = 描述符.value;

  描述符.value = function (...args: any[]) {
    console.log(`调用 ${属性名}，参数：`, args);
    const 结果 = 原方法.apply(this, args);
    console.log(`${属性名} 返回：`, 结果);
    return 结果;
  };
}

// 属性装饰器
function 装饰属性(目标: any, 属性名: string) {
  let 值: any;
  Object.defineProperty(目标, 属性名, {
    get: () => {
      console.log(`读取 ${属性名}`);
      return 值;
    },
    set: (新值: any) => {
      console.log(`设置 ${属性名} = ${新值}`);
      值 = 新值;
    },
  });
}

// 使用
@装饰类
class 餐厅4 {
  @装饰属性 名字: string;

  constructor(名字: string) {
    this.名字 = 名字;
  }

  @装饰方法
  点菜(菜名: string): string {
    return `${this.名字}做${菜名}`;
  }
}
~~~

~~~javascript
// ========== Vue 3 中的装饰器模式（Composition API 风格） ==========

// Vue 3 不依赖装饰器语法，而是用"组合式函数"实现装饰器模式
function useLogger(fn, 名称) {
  return function (...参数) {
    console.log(`[${名称}] 调用`, 参数);
    const 结果 = fn(...参数);
    console.log(`[${名称}] 结果`, 结果);

function useDebounce(fn, 延迟 = 300) {
  let 定时器;
~~~
