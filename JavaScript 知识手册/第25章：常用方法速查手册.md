# 第25章：常用方法速查手册

## 25.1 String 方法

### 查找类

~~~javascript
const 备注 = "不要辣，少放盐，加醋";

// includes：包含某个子串吗？（返回 true/false）
备注.includes("辣")         // true
备注.includes("香菜")       // false

// startsWith：以什么开头？
备注.startsWith("不要")     // true
备注.startsWith("少放")     // false

// endsWith：以什么结尾？
备注.endsWith("加醋")       // true
备注.endsWith("少放盐")     // false

// indexOf：子串第一次出现的位置（找不到返回 -1）
备注.indexOf("少放")        // 4
备注.indexOf("花生")        // -1

// lastIndexOf：子串最后一次出现的位置
"aabbccaa".lastIndexOf("aa")  // 6

// search：支持正则表达式的查找
"今日特价38元".search(/\d+/)  // 4
~~~

### 截取/提取类

~~~javascript
const 菜名 = "招牌秘制红烧肉";

// slice(起始, 结束)：截取一段（不包含结束位置，支持负数）
菜名.slice(4)           // "红烧肉"（从第4个到最后）
菜名.slice(0, 4)        // "招牌秘制"（从第0个到第4个，不包含第4个）
菜名.slice(-3)          // "红烧肉"（倒数3个）

// substring(起始, 结束)：类似 slice，但不支持负数
菜名.substring(0, 4)    // "招牌秘制"

// charAt(索引)：取某个位置的字符
菜名.charAt(0)          // "招"

// at(索引)：ES2022，支持负数索引
菜名.at(0)              // "招"
菜名.at(-1)             // "肉"（最后一个）
~~~

### 转换类

~~~javascript
// toUpperCase / toLowerCase：大小写转换
"Hello".toUpperCase()   // "HELLO"
"HELLO".toLowerCase()   // "hello"

// trim / trimStart / trimEnd：去除空格
"  红烧肉  ".trim()       // "红烧肉"
"  红烧肉  ".trimStart()  // "红烧肉  "
"  红烧肉  ".trimEnd()    // "  红烧肉"

// padStart / padEnd：补全到指定长度
"5".padStart(3, "0")     // "005"
"5".padEnd(3, "0")       // "500"
"VIP".padStart(10, "-")  // "-------VIP"

// repeat：重复
"好吃！".repeat(3)       // "好吃！好吃！好吃！"

// replace / replaceAll：替换
"红烧肉配米饭".replace("米饭", "面条")       // "红烧肉配面条"（只替第一个）
"aaa".replaceAll("a", "b")                    // "bbb"（全部替换）
"价格38元".replace(/\d+/, "**")               // "价格**元"

// split：按分隔符拆成数组
"红烧肉,麻婆豆腐,糖醋里脊".split(",")   // ["红烧肉", "麻婆豆腐", "糖醋里脊"]
"hello world".split(" ")                // ["hello", "world"]
"abc".split("")                          // ["a", "b", "c"]

// match：用正则匹配
"今日特价38元，原价50元".match(/\d+/g)   // ["38", "50"]

// toString：转成字符串
(42).toString()      // "42"
(true).toString()    // "true"
~~~

## 25.2 Number 方法与 Math

~~~javascript
// ========== Number 方法 ==========

// toFixed(位数)：保留几位小数（返回字符串）
(3.14159).toFixed(2)      // "3.14"
(3.14159).toFixed(0)      // "3"

// toPrecision(位数)：保留几位有效数字
(3.14159).toPrecision(4)  // "3.142"
(123.456).toPrecision(2)  // "1.2e+2"

// toString(进制)：转成不同进制的字符串
(255).toString(16)        // "ff"
(10).toString(2)          // "1010"
(255).toString(8)         // "377"

// Number()：转成数字
Number("42")              // 42
Number("3.14")            // 3.14
Number("")                // 0
Number("abc")             // NaN
Number(true)              // 1
Number(false)             // 0
Number(null)              // 0
Number(undefined)         // NaN

// parseInt / parseFloat：从字符串提取数字
parseInt("38元")          // 38
parseFloat("3.14元")      // 3.14
parseInt("ff", 16)        // 255

// isNaN / isFinite / isInteger
isNaN(NaN)                // true
Number.isNaN(NaN)         // true（推荐）
isFinite(42)              // true
Number.isFinite(42)       // true（推荐）
Number.isInteger(42)      // true
Number.isInteger(3.14)    // false

Math.min(10, 30, 5)       // 5（最小值）
Math.sqrt(16)             // 4（开方）

Math.random()             // 0~1随机小数

Math.sign(-5)             // -1（负数返回-1，正数返回1，0返回0）
~~~

## 25.3 Array 方法

### 增删

~~~javascript
const 菜单 = ["红烧肉", "麻婆豆腐", "糖醋里脊"];

// push：从末尾加一个或多个
菜单.push("宫保鸡丁");            // 4（返回新长度）

// pop：从末尾取走一个
菜单.pop();                       // "宫保鸡丁"

// unshift：从开头加一个或多个
菜单.unshift("凉拌黄瓜");

// shift：从开头取走一个
菜单.shift();                     // "凉拌黄瓜"

// splice(起始, 删除数量, ...新增项)：万能增删改
const 原始 = ["红烧肉", "麻婆豆腐", "糖醋里脊", "宫保鸡丁"];
原始.splice(1, 2);               // 删除：["麻婆豆腐", "糖醋里脊"]
原始.splice(1, 0, "鱼香肉丝");   // 插入
原始.splice(1, 1, "水煮鱼");     // 替换

// fill：用某个值填充
new Array(5).fill("空位")           // ["空位", "空位", "空位", "空位", "空位"]
~~~

### 查找类

~~~javascript
const 价格们 = [38, 22, 45, 32, 38];

// indexOf：第一次出现的位置（找不到返回-1）
价格们.indexOf(38)          // 0
价格们.indexOf(99)          // -1

// lastIndexOf：最后一次出现的位置
价格们.lastIndexOf(38)      // 4

// includes：包含某个值吗？
价格们.includes(38)         // true

// find：找到第一个满足条件的元素
const 菜品 = [
  { 名字: "红烧肉", 价格: 38 },
  { 名字: "麻婆豆腐", 价格: 22 },
  { 名字: "糖醋里脊", 价格: 45 }
];
菜品.find(菜 => 菜.价格 > 40)      // { 名字: "糖醋里脊", 价格: 45 }

// findIndex：找到第一个满足条件的索引
菜品.findIndex(菜 => 菜.名字 === "麻婆豆腐")   // 1

// findLast / findLastIndex：ES2023
价格们.findLast(x => x > 30)       // 32
价格们.findLastIndex(x => x > 30)  // 3
~~~

### 遍历/转换类

~~~javascript
const 菜单 = [
  { 名字: "红烧肉", 价格: 38, 类型: "热菜" },
  { 名字: "麻婆豆腐", 价格: 22, 类型: "热菜" },
  { 名字: "拍黄瓜", 价格: 12, 类型: "凉菜" }
];

// map：遍历，返回新数组
菜单.map(菜 => 菜.名字)   // ["红烧肉", "麻婆豆腐", "拍黄瓜"]
菜单.map((菜, i) => `${i + 1}. ${菜.名字}(${菜.价格}元)`)
// ["1. 红烧肉(38元)", "2. 麻婆豆腐(22元)", "3. 拍黄瓜(12元)"]

// filter：过滤，返回满足条件的新数组
菜单.filter(菜 => 菜.类型 === "热菜")    // [{红烧肉}, {麻婆豆腐}]
菜单.filter(菜 => 菜.价格 > 20)          // [{红烧肉}, {麻婆豆腐}]

// reduce：累积，把数组"浓缩"成一个值
菜单.reduce((累加, 菜) => 累加 + 菜.价格, 0)   // 72

// reduce 分组
菜单.reduce((结果, 菜) => {
  if (!结果[菜.类型]) 结果[菜.类型] = [];
  结果[菜.类型].push(菜);
  return 结果;
}, {})
// { 热菜: [{...}, {...}], 凉菜: [{...}] }

// forEach：遍历，没有返回值
菜单.forEach((菜, 索引) => {
  console.log(`${索引}: ${菜.名字}`);
});

// some：有没有任何一个满足条件？
[38, 22, 45].some(x => x > 40)    // true
[38, 22, 45].some(x => x > 100)   // false

// every：是不是全部满足条件？
[38, 22, 45].every(x => x > 0)    // true
[38, 22, 45].every(x => x > 20)   // false

// flat：展平嵌套数组
[1, [2, 3], [4, [5]]].flat()      // [1, 2, 3, 4, [5]]
[1, [2, [3]]].flat(Infinity)      // [1, 2, 3]

// flatMap：先 map 再 flat(1)
["红烧肉 麻婆豆腐", "糖醋里脊"].flatMap(文字 => 文字.split(" "))
// ["红烧肉", "麻婆豆腐", "糖醋里脊"]

// Array.from：把类数组/可迭代对象转成数组
Array.from("红烧肉")              // ["红", "烧", "肉"]
Array.from({ length: 5 }, (_, i) => i + 1)  // [1, 2, 3, 4, 5]

// Array.isArray：判断是不是数组
Array.isArray([1, 2])            // true
Array.isArray("1,2")             // false
~~~

### 排序类

~~~javascript
const 价格们 = [45, 12, 38, 22, 32];

// sort：排序（会改变原数组！）
// 默认按字符串排序（坑！）
[10, 9, 2, 100].sort()            // [10, 100, 2, 9]

// 正确的数字排序
价格们.sort((甲, 乙) => 甲 - 乙)   // [12, 22, 32, 38, 45]（升序）
价格们.sort((甲, 乙) => 乙 - 甲)   // [45, 38, 32, 22, 12]（降序）

// reverse：反转（会改变原数组！）
[1, 2, 3].reverse()              // [3, 2, 1]

// toSorted / toReversed：不改变原数组（ES2023）
const 原 = [3, 1, 2];
原.toSorted()                     // [1, 2, 3]（新数组）
原.toReversed()                   // [2, 1, 3]（新数组）
~~~

### 合并/拆分类

~~~javascript
const A = ["红烧肉", "麻婆豆腐"];
const B = ["糖醋里脊", "宫保鸡丁"];

// concat：合并数组（不改变原数组）
A.concat(B)                       // ["红烧肉", "麻婆豆腐", "糖醋里脊", "宫保鸡丁"]

// 展开运算符（更常用）
[...A, ...B]                      // 同上

// join：数组拼成字符串
A.join("、")                      // "红烧肉、麻婆豆腐"
[2025, 7, 21].join("-")           // "2025-7-21"

// slice(起始, 结束)：截取一段（不改变原数组）
const 所有菜 = ["红烧肉", "麻婆豆腐", "糖醋里脊", "宫保鸡丁", "回锅肉"];
所有菜.slice(1, 3)               // ["麻婆豆腐", "糖醋里脊"]
所有菜.slice(-2)                  // ["宫保鸡丁", "回锅肉"]

// at：ES2022，支持负数索引
所有菜.at(0)                      // "红烧肉"
所有菜.at(-1)                     // "回锅肉"
~~~

## 25.4 Object 方法

~~~javascript
const 菜品 = { 名字: "红烧肉", 价格: 38, 类型: "热菜" };

// keys / values / entries
Object.keys(菜品)                  // ["名字", "价格", "类型"]
Object.values(菜品)                // ["红烧肉", 38, "热菜"]
Object.entries(菜品)
// [["名字", "红烧肉"], ["价格", 38], ["类型", "热菜"]]

// assign / 展开运算符：合并对象
Object.assign({}, { 名字: "红烧肉" }, { 价格: 38 })
// { 名字: "红烧肉", 价格: 38 }
const 合并 = { ...{ 名字: "红烧肉" }, ...{ 价格: 38 } }

// fromEntries：entries 的逆操作
Object.fromEntries([["名字", "红烧肉"], ["价格", 38]])
// { 名字: "红烧肉", 价格: 38 }

// freeze：冻结（不能增删改）
const 固定菜单 = Object.freeze({ 红烧肉: 38 });
固定菜单.红烧肉 = 100;              // 静默失败

// seal：封闭（不能增删，能改现有属性）
const 封闭菜单 = Object.seal({ 红烧肉: 38 });
封闭菜单.红烧肉 = 42;               // ✅
封闭菜单.新菜 = 50;                  // ❌

// hasOwn：ES2022，判断是否是自身属性
Object.hasOwn(菜品, "名字")          // true
Object.hasOwn(菜品, "toString")      // false

// groupBy：ES2024，按条件分组
const 所有菜 = [
  { 名字: "红烧肉", 类型: "热菜" },
  { 名字: "拍黄瓜", 类型: "凉菜" }
];
Object.groupBy(所有菜, 菜 => 菜.类型)
// { 热菜: [{...}], 凉菜: [{...}] }
~~~

## 25.6 Set 方法（含集合运算）

~~~javascript
const 集合 = new Set(["猪肉", "豆腐", "辣椒"]);

// add(值) / has(值) / delete(值) / size / clear
集合.add("花椒");
集合.has("猪肉")          // true
集合.delete("辣椒")       // true
集合.size                // 3

// 数组去重
const 重复 = [1, 2, 2, 3, 3, 3];
[...new Set(重复)]         // [1, 2, 3]

// 集合运算
const A = new Set([1, 2, 3, 4]);
const B = new Set([3, 4, 5, 6]);

// 交集
const 交 = new Set([...A].filter(x => B.has(x)));       // Set { 3, 4 }

// 并集
const 并 = new Set([...A, ...B]);                        // Set { 1,2,3,4,5,6 }

// 差集（A有B没有）
const 差 = new Set([...A].filter(x => !B.has(x)));       // Set { 1, 2 }
~~~

## 25.8 JSON 方法

~~~javascript
const 菜品 = { 名字: "红烧肉", 价格: 38, 配料: ["五花肉", "酱油"] };

// stringify：对象 → 字符串

// '{"名字":"红烧肉","价格":38,"配料":["五花肉","酱油"]}'

  if (键 === "日期") return new Date(值);
  return 值;
});
~~~

## 25.9 Promise 方法

~~~javascript
// 创建
Promise.resolve("成功")             // 立即成功的 Promise
Promise.reject("失败")              // 立即失败的 Promise

// 链式调用
做菜("红烧肉")
  .then(结果 => console.log(结果))   // 成功
  .catch(错误 => console.log(错误))  // 失败
  .finally(() => console.log("结束")) // 无论如何

// 并发控制
Promise.all([做菜A(), 做菜B()])      // 全部成功才成功
  .then(([A, B]) => console.log(A, B))

Promise.allSettled([做菜A(), 做菜B()]) // 等全部完成
  .then(结果们 => 结果们.forEach(r => console.log(r.status)))

Promise.race([请求A(), 请求B()])     // 谁快用谁
  .then(最快的结果 => console.log(结果))

Promise.any([供应商A(), 供应商B()])  // 谁成功用谁
  .then(第一个成功的 => console.log(结果))
~~~

## 25.12 类型转换速查

~~~javascript
// ========== 转数字 ==========
Number("42")             // 42
Number("")               // 0
Number("abc")            // NaN
Number(true)             // 1
Number(null)             // 0
Number(undefined)        // NaN
parseInt("38元")         // 38
parseFloat("3.14元")     // 3.14
+"42"                    // 42（一元加号）

// ========== 转字符串 ==========
String(42)               // "42"
String(true)             // "true"
String(null)             // "null"
(42).toString()          // "42"
(255).toString(16)       // "ff"
`${42}`                  // "42"

// ========== 转布尔 ==========
Boolean(0)               // false
Boolean("")              // false
Boolean(null)            // false
Boolean(undefined)       // false
Boolean(NaN)             // false
Boolean(42)              // true
Boolean("0")             // true ← 坑！
Boolean([])              // true ← 坑！
Boolean({})              // true ← 坑！
!!0                      // false
!!"红烧肉"               // true

// ========== 对象 ↔ 字符串 ==========
JSON.stringify({ a: 1 })  // '{"a":1}'
JSON.parse('{"a":1}')     // { a: 1 }

// ========== 数组 ↔ 字符串 ==========
[1, 2, 3].join(",")       // "1,2,3"
"1,2,3".split(",")        // ["1", "2", "3"]

// ========== 对象 ↔ Map ==========
new Map(Object.entries({ a: 1 }))           // Map { "a" => 1 }
Object.fromEntries(new Map([["a", 1]]))     // { a: 1 }

// ========== 数组 ↔ Set ==========
[...new Set([1, 2, 2, 3])]                 // [1, 2, 3]
Array.from(new Set([1, 2, 3]))             // [1, 2, 3]
~~~

## 25.13 展开运算符

~~~javascript
// ========== 展开数组 ==========
const A = [1, 2, 3];
const B = [4, 5, 6];
const C = [...A, ...B]               // [1, 2, 3, 4, 5, 6]
const D = [...A, 99, ...B]           // [1, 2, 3, 99, 4, 5, 6]

// ========== 展开对象 ==========
const 基础 = { 名字: "红烧肉", 价格: 38 };
const 覆盖 = { ...基础, 价格: 42 }   // { 名字: "红烧肉", 价格: 42 }
const 扩展 = { ...基础, 厨师: "张师傅" } // { 名字: "红烧肉", 价格: 38, 厨师: "张师傅" }

// ========== 收集参数 ==========
function 做菜(主菜, ...配菜们) {
  console.log(主菜, 配菜们);
}
做菜("红烧肉", "米饭", "汤", "水果")
// "红烧肉" ["米饭", "汤", "水果"]

// ========== 浅拷贝 ==========
const 原数组 = [1, 2, 3];
const 拷贝数组 = [...原数组];

const 原对象 = { a: 1 };
const 拷贝对象 = { ...原对象 };
~~~
