## 套路一：数组与哈希表 — 用空间换时间

### 核心思想

~~~
哈希表 = Map / Object / Set
核心作用：把 O(n) 的查找变成 O(1)
代价：多用 O(n) 的空间

一句话：遇到"找某个元素是否存在"的问题 → 先想哈希表
~~~

### 题1：两数之和（LeetCode 1）

~~~javascript
// ========== 题目：给定数组和目标值，找到两个数的下标使它们之和等于目标值
// 输入: nums = [2, 7, 11, 15], target = 9
// 输出: [0, 1]  因为 nums[0] + nums[1] = 2 + 7 = 9

// ========== 思路：
// 暴力法：两层循环 O(n²) → 太慢
// 哈希表法：遍历数组，对于每个数，看"目标值-当前数"是否在哈希表中
// 时间 O(n)，空间 O(n)

// ========== 手写口诀：
// **遍历数组存哈希，键是数值，值是下标**
// **每次看差值是否在表中，在就返回两个下标**

// ========== 代码
function 两数之和(数组, 目标值) {
  // 哈希表：键=数值，值=下标
  const 已见过 = new Map();

  for (let i = 0; i < 数组.length; i++) {
    const 差值 = 目标值 - 数组[i];

    if (已见过.has(差值)) {
      // 找到了！返回两个下标
      return [已见过.get(差值), i];
    }

    // 没找到，把当前数存入哈希表
    已见过.set(数组[i], i);
  }

  return []; // 没有解
}

// ========== 测试
console.log(两数之和([2, 7, 11, 15], 9));   // 输出 ✅ [0, 1]
console.log(两数之和([3, 2, 4], 6));         // 输出 ✅ [1, 2]
console.log(两数之和([3, 3], 6));             // 输出 ✅ [0, 1]
~~~

### 题2：字母异位词分组（LeetCode 49）

~~~javascript
// ========== 题目：把字母相同但顺序不同的词分到一组
// 输入: ["eat","tea","tan","ate","nat","bat"]
// 输出: [["eat","tea","ate"],["tan","nat"],["bat"]]

// ========== 思路：
// 关键：怎么判断两个词是异位词？→ 排序后相同
// 所以用"排序后的字符串"作为哈希表的键

// ========== 手写口诀：
// **排序做键，原词入值，相同键的归为一组**

function 异位词分组(单词列表) {
  const 分组 = new Map();

  for (const 单词 of 单词列表) {
    // 排序后的字符串作为键
    const 键 = 单词.split("").sort().join("");

    if (!分组.has(键)) {
      分组.set(键, []);
    }
    分组.get(键).push(单词);
  }

  return Array.from(分组.values());
}

// ========== 测试
console.log(异位词分组(["eat", "tea", "tan", "ate", "nat", "bat"]));
// 输出 ✅ [["eat","tea","ate"],["tan","nat"],["bat"]]
~~~

### 题3：最长连续序列（LeetCode 128）

~~~javascript
// ========== 题目：找数组中最长的连续递增序列长度（不要求连续位置）
// 输入: [100, 4, 200, 1, 3, 2]
// 输出: 4  （因为 [1, 2, 3, 4] 是最长连续序列）

// ========== 思路：
// 用 Set 去重 + 只从"序列起点"开始计数
// 什么是序列起点？nums[i]-1 不在 Set 中 → 说明它是某段序列的开头

// ========== 手写口诀：
// **数组转集合，只从起点开始数**
// **起点判断：前一个数不在集合中**

function 最长连续序列(数组) {
  const 数字集合 = new Set(数组);
  let 最大长度 = 0;

  for (const 数字 of 数字集合) {
    // 只从起点开始（前一个数不存在）
    if (!数字集合.has(数字 - 1)) {
      let 当前数字 = 数字;
      let 当前长度 = 1;

      // 往后数连续的数
      while (数字集合.has(当前数字 + 1)) {
        当前数字++;
        当前长度++;
      }

      最大长度 = Math.max(最大长度, 当前长度);
    }
  }

  return 最大长度;
}

// ========== 测试
console.log(最长连续序列([100, 4, 200, 1, 3, 2]));  // 输出 ✅ 4
console.log(最长连续序列([0, 3, 7, 2, 5, 8, 4, 6, 0, 1]));  // 输出 ✅ 9
~~~

## 套路二：双指针 — 两个指针协作

### 核心思想

~~~
双指针 = 两个指针从不同位置出发，按照规则移动
两种模式：
  1. 对撞指针：左右两端向中间移动 → 有序数组、回文
  2. 快慢指针：一快一慢 → 链表环检测、找中点
~~~

### 题4：三数之和（LeetCode 15）

~~~javascript
// ========== 题目：找数组中所有和为 0 的三元组（不重复）
// 输入: [-1, 0, 1, 2, -1, -4]
// 输出: [[-1, -1, 2], [-1, 0, 1]]

// ========== 思路：
// 排序 → 固定一个数 → 对剩下部分用双指针找两数之和
// 排序后可以方便地跳过重复元素

// ========== 手写口诀：
// **先排序，外层固定一个数，内层双指针对撞**
// **和大了右指针左移，和小了左指针右移**
// **跳过重复避免答案重复**

function 三数之和(数组) {
  const 结果 = [];
  数组.sort((a, b) => a - b); // 先排序

  for (let i = 0; i < 数组.length - 2; i++) {
    // 跳过重复的第一个数
    if (i > 0 && 数组[i] === 数组[i - 1]) continue;

    let 左 = i + 1;
    let 右 = 数组.length - 1;

    while (左 < 右) {
      const 总和 = 数组[i] + 数组[左] + 数组[右];

      if (总和 === 0) {
        结果.push([数组[i], 数组[左], 数组[右]]);
        // 跳过重复
        while (左 < 右 && 数组[左] === 数组[左 + 1]) 左++;
        while (左 < 右 && 数组[右] === 数组[右 - 1]) 右--;
        左++;
        右--;
      } else if (总和 < 0) {
        左++;   // 和太小，左指针右移（让总和变大）
      } else {
        右--;   // 和太大，右指针左移（让总和变小）
      }
    }
  }

  return 结果;
}

// ========== 测试
console.log(三数之和([-1, 0, 1, 2, -1, -4]));
// 输出 ✅ [[-1, -1, 2], [-1, 0, 1]]
console.log(三数之和([0, 1, 1]));   // 输出 ✅ []
console.log(三数之和([0, 0, 0]));   // 输出 ✅ [[0, 0, 0]]
~~~

### 题5：接雨水（LeetCode 42）

~~~javascript
// ========== 题目：给定柱子高度，计算能接多少雨水
// 输入: [0,1,0,2,1,0,1,3,2,1,2,1]
// 输出: 6

// ========== 思路：
// 每个位置能接的水 = min(左边最高, 右边最高) - 当前高度
// 用双指针从两端向中间走，维护左边最大值和右边最大值

// ========== 手写口诀：
// **左右双指针，维护两侧最大值**
// **哪边小就移动哪边，积水量 = max - 当前高度**

function 接雨水(高度数组) {
  let 左 = 0, 右 = 高度数组.length - 1;
  let 左最大 = 0, 右最大 = 0;
  let 总水量 = 0;

  while (左 < 右) {
    左最大 = Math.max(左最大, 高度数组[左]);
    右最大 = Math.max(右最大, 高度数组[右]);

    if (左最大 < 右最大) {
      // 左边矮，左边能接水
      总水量 += 左最大 - 高度数组[左];
      左++;
    } else {
      // 右边矮，右边能接水
      总水量 += 右最大 - 高度数组[右];
      右--;
    }
  }

  return 总水量;
}

// ========== 测试
console.log(接雨水([0, 1, 0, 2, 1, 0, 1, 3, 2, 1, 2, 1]));  // 输出 ✅ 6
console.log(接雨水([4, 2, 0, 3, 2, 5]));                       // 输出 ✅ 9
~~~

## 套路三：滑动窗口 — 窗口左右滑动

### 核心思想

~~~
滑动窗口 = 一个窗口在数组/字符串上滑动
右指针负责扩大窗口，左指针负责缩小窗口
适用于：找"最长/最短的满足条件的子数组/子串"

模板：
  let left = 0;
  for (let right = 0; right < 数组.length; right++) {
    // 右指针进入窗口 → 更新窗口状态
    while (窗口不满足条件) {
      // 左指针离开窗口 → 缩小窗口
      left++;
    }
    // 更新答案
  }
~~~

### 题6：无重复字符的最长子串（LeetCode 3）

~~~javascript
// ========== 题目：找没有重复字符的最长子串长度
// 输入: "abcabcbb"
// 输出: 3  （"abc"）

// ========== 手写口诀：
// **右指针扩窗口，用 Map 记录字符位置**
// **遇到重复，左指针跳到重复位置的下一个**

function 无重复最长子串(字符串) {
  const 字符位置 = new Map();  // 字符 → 最后出现的位置
  let 最大长度 = 0;
  let 左 = 0;

  for (let 右 = 0; 右 < 字符串.length; 右++) {
    const 当前字符 = 字符串[右];

    // 如果字符重复，左指针跳到重复位置的下一个
    if (字符位置.has(当前字符) && 字符位置.get(当前字符) >= 左) {
      左 = 字符位置.get(当前字符) + 1;
    }

    字符位置.set(当前字符, 右);
    最大长度 = Math.max(最大长度, 右 - 左 + 1);
  }

  return 最大长度;
}

// ========== 测试
console.log(无重复最长子串("abcabcbb"));  // 输出 ✅ 3 ("abc")
console.log(无重复最长子串("bbbbb"));      // 输出 ✅ 1 ("b")
console.log(无重复最长子串("pwwkew"));     // 输出 ✅ 3 ("wke")
~~~

### 题7：找到字符串中所有字母异位词（LeetCode 438）

~~~javascript
// ========== 题目：在字符串 s 中找到所有 p 的异位词的起始下标
// 输入: s = "cbaebabacd", p = "abc"
// 输出: [0, 6]

// ========== 手写口诀：
// **固定大小的窗口滑动，用数组计数代替排序**
// **右进左出，每步比较计数数组是否相等**

function 找异位词(字符串, 模式) {
  const 结果 = [];
  if (模式.length > 字符串.length) return 结果;

  const 模式计数 = new Array(26).fill(0);
  const 窗口计数 = new Array(26).fill(0);
  const 基准 = "a".charCodeAt(0);

  // 初始化模式计数和第一个窗口
  for (let i = 0; i < 模式.length; i++) {
    模式计数[模式.charCodeAt(i) - 基准]++;
    窗口计数[字符串.charCodeAt(i) - 基准]++;
  }

  // 比较函数
  const 相等 = () => 模式计数.every((v, i) => v === 窗口计数[i]);
  if (相等()) 结果.push(0);

  // 滑动窗口
  for (let 右 = 模式.length; 右 < 字符串.length; 右++) {
    // 右边进入
    窗口计数[字符串.charCodeAt(右) - 基准]++;
    // 左边离开
    窗口计数[字符串.charCodeAt(右 - 模式.length) - 基准]--;

    if (相等()) 结果.push(右 - 模式.length + 1);
  }

  return 结果;
}

// ========== 测试
console.log(找异位词("cbaebabacd", "abc"));  // 输出 ✅ [0, 6]
console.log(找异位词("abab", "ab"));          // 输出 ✅ [0, 1, 2]
~~~

## 套路四：链表 — 指针操作

### 核心思想

~~~
链表题的核心：画图 + 操作指针
关键技巧：
  1. 虚拟头节点（dummy）→ 不用特殊处理头节点
  2. 快慢指针 → 找中点、检测环
  3. 先存 next 再修改 → 防止断链
~~~

### 链表基础定义

~~~javascript
// ========== 链表节点定义
class ListNode {
  constructor(值, 下一个 = null) {
    this.val = 值;
    this.next = 下一个;
  }
}

// 工具函数：数组转链表
function 数组转链表(数组) {
  const 虚拟头 = new ListNode(0);
  let 当前 = 虚拟头;
  for (const 值 of 数组) {
    当前.next = new ListNode(值);
    当前 = 当前.next;
  }
  return 虚拟头.next;
}

// 工具函数：链表转数组（方便测试）
function 链表转数组(头节点) {
  const 结果 = [];
  let 当前 = 头节点;
  while (当前) {
    结果.push(当前.val);
    当前 = 当前.next;
  }
  return 结果;
}
~~~

### 题8：反转链表（LeetCode 206）

~~~javascript
// ========== 题目：反转一个单链表
// 输入: 1 → 2 → 3 → 4 → 5
// 输出: 5 → 4 → 3 → 2 → 1

// ========== 手写口诀：
// **三指针法：前一个、当前、下一个**
// **先把 next 存起来，再把当前指向前一个，然后三个指针都往前走**

function 反转链表(头节点) {
  let 前一个 = null;
  let 当前 = 头节点;

  while (当前) {
    const 下一个 = 当前.next;  // 1. 先存下一个
    当前.next = 前一个;         // 2. 反转：当前指向前一个
    前一个 = 当前;              // 3. 前一个往前走
    当前 = 下一个;              // 4. 当前往前走
  }

  return 前一个; // 前一个就是新的头节点
}

// ========== 测试
const 链表1 = 数组转链表([1, 2, 3, 4, 5]);
console.log(链表转数组(反转链表(链表1)));  // 输出 ✅ [5, 4, 3, 2, 1]
~~~

### 题9：合并两个有序链表（LeetCode 21）

~~~javascript
// ========== 题目：合并两个有序链表为一个有序链表
// 输入: 1→2→4, 1→3→4
// 输出: 1→1→2→3→4→4

// ========== 手写口诀：
// **虚拟头节点，比较两个链表的头，小的接到结果后面**

function 合并有序链表(链表1, 链表2) {
  const 虚拟头 = new ListNode(0);
  let 当前 = 虚拟头;

  while (链表1 && 链表2) {
    if (链表1.val <= 链表2.val) {
      当前.next = 链表1;
      链表1 = 链表1.next;
    } else {
      当前.next = 链表2;
      链表2 = 链表2.next;
    }
    当前 = 当前.next;
  }

  // 接上剩余部分
  当前.next = 链表1 || 链表2;

  return 虚拟头.next;
}

// ========== 测试
const a = 数组转链表([1, 2, 4]);
const b = 数组转链表([1, 3, 4]);
console.log(链表转数组(合并有序链表(a, b)));  // 输出 ✅ [1, 1, 2, 3, 4, 4]
~~~

### 题10：环形链表（LeetCode 141）

~~~javascript
// ========== 题目：判断链表是否有环

// ========== 手写口诀：
// **快慢指针，快走两步慢走一步，相遇就有环**

function 有环吗(头节点) {
  let 慢 = 头节点;
  let 快 = 头节点;

  while (快 && 快.next) {
    慢 = 慢.next;          // 慢指针走一步
    快 = 快.next.next;     // 快指针走两步

    if (慢 === 快) return true;  // 相遇 → 有环
  }

  return false; // 快指针到头了 → 没环
}

// ========== 测试
const 无环链表 = 数组转链表([1, 2, 3, 4]);
console.log(有环吗(无环链表));  // 输出 ✅ false

// 构造有环链表：1 → 2 → 3 → 4 → 回到 2
const 有环头 = new ListNode(1);
有环头.next = new ListNode(2);
有环头.next.next = new ListNode(3);
有环头.next.next.next = new ListNode(4);
有环头.next.next.next.next = 有环头.next; // 4 → 2 形成环
console.log(有环吗(有环头));  // 输出 ✅ true
~~~

## 套路五：栈与队列

### 核心思想

~~~
栈：后进先出（LIFO）→ 函数调用栈、撤销操作
队列：先进先出（FIFO）→ BFS 层序遍历
单调栈：栈中元素保持单调递增/递减 → 找"下一个更大/更小的元素"
~~~

### 题11：有效括号（LeetCode 20）

~~~javascript
// ========== 题目：判断括号字符串是否有效
// 输入: "()" → true, "()[]{}" → true, "(]" → false

// ========== 手写口诀：
// **遇到左括号压栈，遇到右括号弹栈匹配**
// **最后栈空才是有效**

function 有效括号(字符串) {
  const 栈 = [];
  const 匹配表 = { ")": "(", "]": "[", "}": "{" };

  for (const 字符 of 字符串) {
    if ("({[".includes(字符)) {
      栈.push(字符);  // 左括号压栈
    } else {
      // 右括号 → 弹栈匹配
      if (栈.length === 0 || 栈.pop() !== 匹配表[字符]) {
        return false;
      }
    }
  }

  return 栈.length === 0; // 栈空才有效
}

// ========== 测试
console.log(有效括号("()"));        // 输出 ✅ true
console.log(有效括号("()[]{}"));    // 输出 ✅ true
console.log(有效括号("(]"));        // 输出 ✅ false
console.log(有效括号("([)]"));      // 输出 ✅ false
console.log(有效括号("{[]}"));      // 输出 ✅ true
~~~

### 题12：最小栈（LeetCode 155）

~~~javascript
// ========== 题目：实现一个栈，能在 O(1) 时间内返回最小值

// ========== 手写口诀：
// **双栈法：一个正常栈，一个最小栈**
// **每次 push 同时更新最小栈的栈顶**

class 最小栈 {
  constructor() {
    this.栈 = [];
    this.最小栈 = [];  // 同步记录每个状态的最小值
  }

  push(值) {
    this.栈.push(值);
    // 最小栈：如果新值更小就压入新值，否则重复压入当前最小值
    const 当前最小 = this.最小栈.length === 0
      ? 值
      : Math.min(值, this.最小栈[this.最小栈.length - 1]);
    this.最小栈.push(当前最小);
  }

  pop() {
    this.栈.pop();
    this.最小栈.pop();
  }

  top() {
    return this.栈[this.栈.length - 1];
  }

  getMin() {
    return this.最小栈[this.最小栈.length - 1];
  }
}

// ========== 测试
const 栈 = new 最小栈();
栈.push(-2);
栈.push(0);
栈.push(-3);
console.log(栈.getMin());  // 输出 ✅ -3
栈.pop();
console.log(栈.top());     // 输出 ✅ 0
console.log(栈.getMin());  // 输出 ✅ -2
~~~

### 题13：每日温度（LeetCode 739）

~~~javascript
// ========== 题目：对于每天的温度，找到下一次更高温度要等几天
// 输入: [73, 74, 75, 71, 69, 72, 76, 73]
// 输出: [1,  1,  4,  2,  1,  1,  0,  0]

// ========== 思路：单调递减栈
// 栈中存下标，保持"温度从栈底到栈顶递减"
// 遇到更高温度时，弹栈并计算天数差

// ========== 手写口诀：
// **单调递减栈，存下标不存温度**
// **当前温度比栈顶高 → 弹栈，天数 = 当前下标 - 栈顶下标**

function 每日温度(温度数组) {
  const 结果 = new Array(温度数组.length).fill(0);
  const 栈 = []; // 存下标，栈底到栈顶温度递减

  for (let i = 0; i < 温度数组.length; i++) {
    // 当前温度比栈顶高 → 弹栈并记录天数
    while (栈.length > 0 && 温度数组[i] > 温度数组[栈[栈.length - 1]]) {
      const 栈顶下标 = 栈.pop();
      结果[栈顶下标] = i - 栈顶下标;
    }
    栈.push(i); // 当前下标入栈
  }

  return 结果;
}

// ========== 测试
console.log(每日温度([73, 74, 75, 71, 69, 72, 76, 73]));
// 输出 ✅ [1, 1, 4, 2, 1, 1, 0, 0]
console.log(每日温度([30, 40, 50, 60]));
// 输出 ✅ [1, 1, 1, 0]
~~~

## 套路六：二叉树 — DFS / BFS

### 核心思想

~~~
二叉树的三种遍历（DFS）：
  前序：根 → 左 → 右
  中序：左 → 根 → 右（二叉搜索树中序 = 有序）
  后序：左 → 右 → 根

BFS（层序遍历）：一层一层从左到右

递归三板斧：
  1. 递归函数的定义是什么？（它能干什么）
  2. 递归终止条件是什么？（什么时候停下来）
  3. 当前层做什么？（缩小问题规模）
~~~

~~~javascript
    this.left = 左;

  }
~~~

### 题16：二叉树的层序遍历（LeetCode 102）

~~~javascript
// ========== 题目：按层从左到右遍历二叉树
// 输出: [[3], [9, 20], [15, 7]]

// ========== 手写口诀：
// **BFS 用队列，每次处理一层**
// **记住当前层的长度，循环这么多次**

function 层序遍历(根节点) {
  if (!根节点) return [];

  const 结果 = [];
  const 队列 = [根节点];

  while (队列.length > 0) {
    const 当前层 = [];
    const 当前层长度 = 队列.length;

    for (let i = 0; i < 当前层长度; i++) {
      const 节点 = 队列.shift();
      当前层.push(节点.val);

      if (节点.left) 队列.push(节点.left);
      if (节点.right) 队列.push(节点.right);
    }

    结果.push(当前层);
  }

  return 结果;
}

// ========== 测试
console.log(层序遍历(树));
// 输出 ✅ [[3], [9, 20], [15, 7]]
~~~

### 题17：验证二叉搜索树（LeetCode 98）

~~~javascript
// ========== 题目：判断是否是有效的二叉搜索树（BST）
// BST 特性：左子树所有节点 < 根 < 右子树所有节点

// ========== 手写口诀：
// **中序遍历 BST 应该是严格递增的**
// **递归传上下界，左子树上界是父节点，右子树下界是父节点**

function 是有效BST(节点, 最小值 = -Infinity, 最大值 = Infinity) {
  if (!节点) return true;

  if (节点.val <= 最小值 || 节点.val >= 最大值) return false;

  // 左子树：上界更新为当前节点值
  // 右子树：下界更新为当前节点值
  return 是有效BST(节点.left, 最小值, 节点.val)
      && 是有效BST(节点.right, 节点.val, 最大值);
}

// ========== 测试
const 有效BST = new TreeNode(2, new TreeNode(1), new TreeNode(3));
console.log(是有效BST(有效BST));  // 输出 ✅ true

const 无效BST = new TreeNode(5,
  new TreeNode(1),
  new TreeNode(4, new TreeNode(3), new TreeNode(6))
);
console.log(是有效BST(无效BST));  // 输出 ✅ false（4 < 5 但 6 > 5 在右子树）
~~~

## 套路七：动态规划入门 — 递推公式

### 核心思想

~~~
动态规划 = 把大问题拆成小问题，从小到大逐步求解
核心步骤：
  1. 定义状态：dp[i] 代表什么？
  2. 写转移方程：dp[i] 和 dp[i-1]、dp[i-2]... 的关系
  3. 确定初始值：dp[0]、dp[1] 是什么
  4. 确定遍历顺序：从小到大

一句话：动态规划就是"填表格"，每个格子的答案由之前的格子推出
~~~

### 题18：爬楼梯（LeetCode 70）

~~~javascript
// ========== 题目：爬 n 级台阶，每次可爬 1 或 2 级，有多少种方法？
// 输入: 3 → 输出: 3（1+1+1, 1+2, 2+1）

// ========== 思路：
// 到第 n 级的方法数 = 到第 n-1 级 + 到第 n-2 级
// 因为最后一步要么走 1 级（从 n-1），要么走 2 级（从 n-2）
// dp[i] = dp[i-1] + dp[i-2] → 就是斐波那契数列！

// ========== 手写口诀：
// **dp[i] = dp[i-1] + dp[i-2]**
// **初始值 dp[1]=1, dp[2]=2**

function 爬楼梯(n) {
  if (n <= 2) return n;

  let 前两个 = 1; // dp[1]
  let 前一个 = 2; // dp[2]

  for (let i = 3; i <= n; i++) {
    const 当前 = 前一个 + 前两个;
    前两个 = 前一个;
    前一个 = 当前;
  }

  return 前一个;
}

// ========== 测试
console.log(爬楼梯(2));   // 输出 ✅ 2
console.log(爬楼梯(3));   // 输出 ✅ 3
console.log(爬楼梯(10));  // 输出 ✅ 89
~~~

### 题19：最大子数组和（LeetCode 53）

~~~javascript
// ========== 题目：找连续子数组的最大和
// 输入: [-2,1,-3,4,-1,2,1,-5,4]
// 输出: 6  （[4,-1,2,1]）

// ========== 思路：
// dp[i] = 以 nums[i] 结尾的最大子数组和
// dp[i] = max(nums[i], dp[i-1] + nums[i])
// 意思：要么自己新开一段，要么接在前面的后面

// ========== 手写口诀：
// **dp[i] = max(自己, 前面的最大和 + 自己)**
// **优化：只用一个变量记录前面的最大和**

function 最大子数组和(数组) {
  let 当前最大 = 数组[0];
  let 全局最大 = 数组[0];

  for (let i = 1; i < 数组.length; i++) {
    当前最大 = Math.max(数组[i], 当前最大 + 数组[i]);
    全局最大 = Math.max(全局最大, 当前最大);
  }

  return 全局最大;
}

// ========== 测试
console.log(最大子数组和([-2, 1, -3, 4, -1, 2, 1, -5, 4]));  // 输出 ✅ 6
console.log(最大子数组和([1]));                                   // 输出 ✅ 1
console.log(最大子数组和([-1, -2]));                              // 输出 ✅ -1
~~~

### 题20：零钱兑换（LeetCode 322）

~~~javascript
// ========== 题目：用最少的硬币数凑出目标金额
// 输入: coins = [1, 2, 5], amount = 11
// 输出: 3  （5 + 5 + 1）

// ========== 思路：
// dp[i] = 凑出金额 i 需要的最少硬币数
// dp[i] = min(dp[i - coin] + 1) for each coin
// 意思：对于每种硬币，"用它"比"不用它"哪个更优

// ========== 手写口诀：
// **dp[i] = 所有 dp[i-coin] 中最小的 + 1**
// **初始 dp[0]=0, 其他初始化为 Infinity**

function 零钱兑换(硬币们, 目标金额) {
  const dp = new Array(目标金额 + 1).fill(Infinity);
  dp[0] = 0; // 金额0需要0个硬币

  for (let 金额 = 1; 金额 <= 目标金额; 金额++) {
    for (const 硬币 of 硬币们) {
      if (金额 >= 硬币 && dp[金额 - 硬币] !== Infinity) {
        dp[金额] = Math.min(dp[金额], dp[金额 - 硬币] + 1);
      }
    }
  }

  return dp[目标金额] === Infinity ? -1 : dp[目标金额];
}

// ========== 测试
console.log(零钱兑换([1, 2, 5], 11));   // 输出 ✅ 3
console.log(零钱兑换([2], 3));            // 输出 ✅ -1（无法凑出）
console.log(零钱兑换([1], 0));            // 输出 ✅ 0
~~~

### 题23：排序算法手写

~~~javascript
// ========== 冒泡排序（了解即可，面试不常问）
function 冒泡排序(数组) {
  const 结果 = [...数组];
  for (let i = 0; i < 结果.length - 1; i++) {
    for (let j = 0; j < 结果.length - 1 - i; j++) {
      if (结果[j] > 结果[j + 1]) {
        [结果[j], 结果[j + 1]] = [结果[j + 1], 结果[j]];
      }
    }
  }
  return 结果;
}

// ========== 快速排序（高频！）
// 手写口诀：**选基准，小的放左大的放右，递归排左右**
function 快速排序(数组) {
  if (数组.length <= 1) return 数组;

  const 基准 = 数组[0];
  const 左边 = [];
  const 右边 = [];

  for (let i = 1; i < 数组.length; i++) {
    if (数组[i] < 基准) {
      左边.push(数组[i]);
    } else {
      右边.push(数组[i]);
    }
  }

  return [...快速排序(左边), 基准, ...快速排序(右边)];
}

// ========== 归并排序（高频！）
// 手写口诀：**分两半，各自排，合并时双指针比较**
function 归并排序(数组) {
  if (数组.length <= 1) return 数组;

  const 中间 = Math.floor(数组.length / 2);
  const 左边 = 归并排序(数组.slice(0, 中间));
  const 右边 = 归并排序(数组.slice(中间));

  // 合并两个有序数组
  const 结果 = [];
  let i = 0, j = 0;
  while (i < 左边.length && j < 右边.length) {
    if (左边[i] <= 右边[j]) {
      结果.push(左边[i++]);
    } else {
      结果.push(右边[j++]);
    }
  }
  return 结果.concat(左边.slice(i)).concat(右边.slice(j));
}

// ========== 测试
console.log(冒泡排序([5, 3, 8, 1, 2]));  // 输出 ✅ [1, 2, 3, 5, 8]
console.log(快速排序([5, 3, 8, 1, 2]));  // 输出 ✅ [1, 2, 3, 5, 8]
console.log(归并排序([5, 3, 8, 1, 2]));  // 输出 ✅ [1, 2, 3, 5, 8]
~~~

### 题24：二分查找

~~~javascript
// ========== 题目：在有序数组中查找目标值的下标
// 输入: [1, 3, 5, 7, 9], target = 5 → 输出: 2

// ========== 手写口诀：
// **左右两个指针，算中间值**
// **中间值等于目标 → 返回**
// **中间值小于目标 → 左指针右移（去右边找）**
// **中间值大于目标 → 右指针左移（去左边找）**

function 二分查找(数组, 目标值) {
  let 左 = 0;
  let 右 = 数组.length - 1;

  while (左 <= 右) {
    const 中 = Math.floor((左 + 右) / 2);

    if (数组[中] === 目标值) {
      return 中;           // 找到了
    } else if (数组[中] < 目标值) {
      左 = 中 + 1;         // 目标在右边
    } else {
      右 = 中 - 1;         // 目标在左边
    }
  }

  return -1; // 没找到
}

// ========== 测试
console.log(二分查找([1, 3, 5, 7, 9], 5));  // 输出 ✅ 2
console.log(二分查找([1, 3, 5, 7, 9], 3));  // 输出 ✅ 1
console.log(二分查找([1, 3, 5, 7, 9], 6));  // 输出 ✅ -1
~~~

## 终极记忆卡片

1. **哈希表** → 遇到"查找/去重/计数" → 先想 Map/Set，空间换时间
2. **双指针** → 有序数组/链表 → 左右对撞或快慢同走
3. **滑动窗口** → 子数组/子串问题 → 右扩左缩，维护窗口状态
4. **链表** → 虚拟头节点防特殊处理，先存 next 再修改防断链
5. **单调栈** → 找"下一个更大/更小元素" → 栈中保持单调
6. **二叉树递归** → 三板斧：定义、终止、当前层做什么
7. **BFS 层序** → 队列 + 记录当前层长度
8. **动态规划** → 定义 dp[i] 的含义 → 写转移方程 → 确定初始值和遍历顺序
9. **排序** → 快排（选基准分左右）、归并（分两半各自排再合并）
10. **二分** → 有序数组查找，左右指针算中间值
